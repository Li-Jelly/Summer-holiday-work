# 算法合集
## 贪心（Greedy）
### 算什么
**最优化问题**（求最大/最小、最多/最少）
### 核心思想
每一步都选**当前看起来最优的**，最终得到全局最优
### 前提
问题具有**最优子结构**
#### 什么是最优子结构
+ **定义**：一个问题的最优解包含**子问题**的最优解
+ **通俗理解**：盖10层楼的最优设计，下面9层也一定是最优设计，才能保证整顿楼最优
+ **数学表达**：问题的最优解=f(子问题1，子问题2，...)，且子问题的解也必须是最优的，就称具有最优子结构
+ **总结**：**局部最优能导致全局最优**
### 代码特征
**排序**+**一次遍历**，没有复杂数据结构
### 如何判断是否能用贪心
**剪掉最后一步**，还是最优解（不是的话只能暴力搜索）
### 典型场景
+ 区间调度（选最多不重叠区间）
+ 活动安排
+ 零钱兑换（特定币值下）
+ 加油站问题
### 题型讲解
#### 区间调度
##### 问题本质
给一堆区间[start,end]，选择**最多数量的不重叠区间**
##### 贪心策略
按**结束时间升序**排序，每次选结束最早的，然后跳过与其重叠的区间
##### 为什么正确
**结束越早，留给后面的空间越多**，典型的最优子结构
##### 例题1
**题目**：有一堆课，同一时间只能上一门，最多上几门课
          输入：[[1,2],[2,3],[3,4],[1,3]]
          输出：3（选[1,2],[2,3],[3,4]）
```
#include<bits/stdc++.h>
using namespace std;

int maxCourses(vector<vector<int>>&courses){//模板类，表示元素也是“vector<int>”的向量，即二维动态数组
                                            //&左值引用，表示courses是传入实参的别名，不会赋值整个二维数组，函数内对courses的修改会直接影响原始数据
  if(courses.empty())
    return 0;

  sort(courses.begin(),courses.end(),       //库函数，排序（迭代器.courses的第一个元素，最后一个元素）“，”参数分隔
    [](const vertor<int>&a,const vector<int>&b){//Lambda表达式的捕获列表，空的表示不捕获外部任何变量，a常量引用，指向一个vector<int>，表示待比较的第一个课程数据，b是第二个
        return a[1]<b[1];                   //比较逻辑：返回ture,如果a的第二个元素（索引1）小于b的第二个元素（索引1）
    });                                     //作用：按照每个向量元素的第二个元素进行升序排序
  int count=1;
  int last_end=courses[0][1];

  for(int i=1;i<courses.size();i++){
    if(courses[i][0]>=last_end){
      count++;
      last_end=courses[i][1];
    }
  }
  return count;
}

int main(){
  vector<vector<int>> courses={{1,2},{2,3,},{3,4},{1,3}};
  cout<<"最多能上"<<maxCourses(courses)<<"门课"<<endl;
  return 0;
}

```

#### 零钱兑换（贪心版本）
##### 例题1
```
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

// 贪心求最少硬币数（仅适用于币值符合贪心条件的场景）
int coinChangeGreedy(vector<int>& coins, int amount) {
    // 先按面值从大到小排序（贪心策略：先用大面值）
    sort(coins.begin(), coins.end(), greater<int>());
    // greater<int>() 表示降序排列，即从大到小
    
    int count = 0;          // 硬币总数
    int remaining = amount; // 剩余未凑金额
    
    // 遍历每种面值的硬币
    for (int coin : coins) {
        // 如果已经凑完，提前退出循环
        if (remaining == 0) {
            break;
        }
        
        // 当前面值最多能用多少个（不能超过剩余金额）
        int num = remaining / coin;  // 整除，取整数部分
        
        if (num > 0) {
            count += num;              // 累加硬币数量
            remaining -= num * coin;   // 减去已凑金额
            
            // 打印使用情况（方便观察过程）
            cout << "使用 " << num << " 个 " << coin << " 元硬币" << endl;
        }
    }
    
    // 如果剩余金额为0，说明成功凑出，返回硬币总数
    // 否则返回-1表示无法凑出
    if (remaining == 0) {
        return count;
    } else {
        return -1;
    }
}

int main() {
    vector<int> coins = {1, 5, 10, 20, 50, 100};
    int amount = 666;
    
    int result = coinChangeGreedy(coins, amount);
    cout << "最少硬币数: " << result << endl;
    
    // 输出：
    // 使用 6 个 100 元硬币
    // 使用 1 个 50 元硬币
    // 使用 1 个 10 元硬币
    // 使用 1 个 5 元硬币
    // 使用 1 个 1 元硬币
    // 最少硬币数: 10
    
    return 0;
}
```
#### 加油站问题
##### 例题
```
#include <iostream>
#include <vector>
using namespace std;

// 返回能绕一圈的加油站索引，如果不存在则返回 -1
int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
    int total_gas = 0;    // 总油量 - 总消耗（判断全局可行性）
    int current_gas = 0;  // 当前油箱剩余油量（用于找起点）
    int start_index = 0;  // 假设起点为0（第一个加油站）
    
    // 遍历每个加油站
    for (int i = 0; i < gas.size(); i++) {
        // 计算到下一个站的净收益
        // 正数说明油变多，负数说明油变少
        int net = gas[i] - cost[i];
        
        total_gas += net;      // 累加总净收益
        current_gas += net;    // 累加当前段的净收益
        
        // 如果当前油量 < 0，说明从 start_index 无法到达 i+1
        if (current_gas < 0) {
            // 将起点设为下一个站（i+1）
            start_index = i + 1;
            // 重置当前油量，从新起点重新开始计算
            current_gas = 0;
            // 注意：total_gas 不重置，它用于全局判断
        }
    }
    
    // 如果总净收益 >= 0，说明存在解，返回 start_index
    // 否则返回 -1
    if (total_gas >= 0) {
        return start_index;
    } else {
        return -1;
    }
}

int main() {
    vector<int> gas = {1, 2, 3, 4, 5};
    vector<int> cost = {3, 4, 5, 1, 2};
    
    int result = canCompleteCircuit(gas, cost);
    cout << "从第 " << result << " 个加油站出发" << endl;
    // 输出：从第 3 个加油站出发（索引从0开始）
    
    // 验证：从索引3（第4个站）出发
    // 到站4：油 = 4 - 1 + 5 = 8
    // 到站0：油 = 8 - 2 + 1 = 7
    // 到站1：油 = 7 - 3 + 2 = 6
    // 到站2：油 = 6 - 4 + 3 = 5
    // 到站3：油 = 5 - 5 + 4 = 4 > 0 成功！
    
    return 0;
}
```
